import java.util.ArrayDeque;
import java.util.Deque;
import java.util.Queue;
import java.util.Scanner;

public class stack_usingQueue {
    int Maxsize;
    Queue<Integer> queue= new ArrayDeque<>();
    void create_stack(int size) {
        Maxsize = size;
    }
    void push(int e){
        queue.add(e);
        System.out.println(e+ "pushed successfully");
    }
    int pop(){
        if(queue.isEmpty()){
            System.out.print("stack is empty!");
            return -1;
        }for (int i = 0; i < queue.size() - 1; i++) {
            queue.add(queue.remove());
        }
        int temp = queue.peek();
        queue.remove();
        return temp;
    }
    int peek() {
        if (queue.isEmpty()) {
            System.out.println("Stack is empty!");
            return -1;
        }

        for (int i = 0; i < queue.size() - 1; i++) {
            queue.add(queue.remove());
        }
        int temp = queue.peek();
        queue.add(queue.remove()); // restore
        return temp;
    }

    boolean is_empty() {
        return queue.isEmpty();
    }

    boolean is_full() {
        return Maxsize == queue.size();
    }

    void print_stack() {
        if (queue.isEmpty()) {
            System.out.println("Stack is empty!");
            return;
        }

        Queue<Integer> temp = new ArrayDeque<>(queue);
        System.out.print("Stack contents (top → bottom): ");
        for (int val : temp) {
            System.out.print(val + " ");
        }
        System.out.println();
    }
}

 class Stackque {

                case 1:
                    if (!obj.is_full()) {
                        System.out.print("Enter number: ");

                        obj.push(e);
                    } else {
                        System.out.println("Stack is full");

                    } else {
                        System.out.println("Stack is empty");
                    }
                    break;

                    if (!obj.is_empty()) {
                        System.out.println(obj.peek());

import java.util.ArrayDeque;
import java.util.Deque;
import java.util.Scanner;

public class Queue_UsingStack {
    Deque<Integer> stack = new ArrayDeque<>();
    Deque<Integer> Auxilary = new ArrayDeque<>();

    void enqueue(int e) {
        stack.push(e);
        System.out.println("enqueued into queue " + e);

    }

    int dequeue() {
        if (stack.isEmpty() && Auxilary.isEmpty()){
            System.out.println("stack is empty! cannot dequeued");
        return -1;
    }
        if(Auxilary.isEmpty()) {
            while (!stack.isEmpty()) {
                Auxilary.push(stack.pop());
            }
        }
        int removed = Auxilary.pop();
        System.out.println(removed + " dequeued from queue: ");
        return removed;
    }

    boolean isEmpty() {
        return stack.isEmpty() && Auxilary.isEmpty();
    }

    //print queue in fifo order
    void print_Queue() {
        if (isEmpty()) {
            System.out.println("queue is empty");
            return;
        }
        //move all to Auxilary to print in correct order
        if (Auxilary.isEmpty()) {
            while (!stack.isEmpty()) {
                Auxilary.push(stack.pop());
            }

        }
        System.out.println("queue element front t rear");
        for (int ele : Auxilary) {
            System.out.println(ele + "");
        }
        System.out.println();
    }
